# Isolated geometry-aware ANN research prototype.
cmake_minimum_required(VERSION 3.18)
project(geometry_ann LANGUAGES CXX)
add_library(geometry_ann index.cpp training.cpp geometry.cpp search_state.cpp search_hierarchy.cpp graph.cpp search_graph.cpp scan_distance.cpp sq8_distance.cpp geometry_diagnostics.cpp support_geometry.cpp cell_bounds.cpp cell_diagnostics.cpp rank_scores.cpp scan_policies.cpp)
find_package(Threads REQUIRED)
target_link_libraries(geometry_ann PUBLIC Threads::Threads)
target_compile_features(geometry_ann PUBLIC cxx_std_20)
target_include_directories(geometry_ann PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(geometry_ann PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
add_executable(geometry_bench benchmark.cpp benchmark_options.cpp dataset.cpp)
target_link_libraries(geometry_bench PRIVATE geometry_ann)
add_executable(geometry_tests tests.cpp graph_tests.cpp graph_settings.cpp)
target_link_libraries(geometry_tests PRIVATE geometry_ann)
target_compile_options(geometry_tests PRIVATE -ffp-contract=off)
enable_testing()
add_test(NAME geometry_tests COMMAND geometry_tests)

add_executable(graph_bench graph_benchmark.cpp graph_evaluation.cpp graph_settings.cpp graph_measurement.cpp benchmark_options.cpp dataset.cpp)
target_link_libraries(graph_bench PRIVATE geometry_ann)

option(GEOMETRY_FAST_SCAN "Use SIMD FP32 rejection before exact double scoring" ON)
if(GEOMETRY_FAST_SCAN)
  target_compile_definitions(geometry_ann PRIVATE GEOMETRY_FAST_SCAN=1)
endif()

add_executable(dataset_tests dataset_tests.cpp dataset.cpp)
target_link_libraries(dataset_tests PRIVATE geometry_ann)
add_test(NAME dataset_tests COMMAND dataset_tests)

add_executable(sq8_oracle sq8_oracle.cpp benchmark_options.cpp dataset.cpp)
target_link_libraries(sq8_oracle PRIVATE geometry_ann)

add_executable(cell_ceiling cell_ceiling.cpp benchmark_options.cpp dataset.cpp)
target_link_libraries(cell_ceiling PRIVATE geometry_ann)
add_executable(cell_tests cell_tests.cpp)
target_link_libraries(cell_tests PRIVATE geometry_ann)
add_test(NAME cell_tests COMMAND cell_tests)
