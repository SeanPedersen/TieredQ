cmake_minimum_required(VERSION 3.18)
project(routerduel LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(TIEREDQ_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/tieredq" CACHE PATH "TieredQ source tree")
add_subdirectory(${TIEREDQ_SOURCE_DIR} tieredq EXCLUDE_FROM_ALL)
find_package(OpenMP REQUIRED)
set(BLA_VENDOR OpenBLAS)
find_package(BLAS REQUIRED)
find_path(CBLAS_INCLUDE_DIR cblas.h PATH_SUFFIXES openblas)
add_executable(duel duel.cpp)
target_include_directories(duel PRIVATE ${CBLAS_INCLUDE_DIR})
target_compile_options(duel PRIVATE -O3 -march=native)
target_link_libraries(duel PRIVATE TieredQ::TieredQ BLAS::BLAS OpenMP::OpenMP_CXX)
