# Isolated experimental bindings; production codec and Python API remain unchanged.
cmake_minimum_required(VERSION 3.18)
project(tiny_budget_experiment LANGUAGES CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_STANDARD 20)
add_compile_options(-O3 -march=native -ffp-contract=off)
set(TIEREDQ_TESTS OFF CACHE BOOL "" FORCE)
set(TIEREDQ_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(../../.. tieredq)
find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
execute_process(COMMAND "${Python_EXECUTABLE}" -m nanobind --cmake_dir OUTPUT_VARIABLE NB_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
list(APPEND CMAKE_PREFIX_PATH "${NB_DIR}")
find_package(nanobind CONFIG REQUIRED)
nanobind_add_module(tiny_native NOMINSIZE ../native/module.cpp)
target_compile_definitions(tiny_native PRIVATE EXPERIMENT_MODULE=tiny_native)
target_link_libraries(tiny_native PRIVATE TieredQ::TieredQ)

option(EXPERIMENT_PROFILE "Instrument a separate diagnostic source snapshot" OFF)
if(EXPERIMENT_PROFILE)
  target_compile_definitions(tiny_native PRIVATE EXPERIMENT_PROFILE=1)
endif()
