cmake_minimum_required(VERSION 3.18)
project(kernel_bench LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(TIEREDQ_SOURCE_DIR "" CACHE PATH "TieredQ source tree (the study's snapshot)")
option(TIEREDQ_TILES "Also time tile scans (B only)" OFF)
set(TIEREDQ_TESTS OFF CACHE BOOL "" FORCE)
set(TIEREDQ_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(${TIEREDQ_SOURCE_DIR} tieredq EXCLUDE_FROM_ALL)
add_executable(kernel_bench ${CMAKE_CURRENT_SOURCE_DIR}/../kernel_bench.cpp)
target_compile_options(kernel_bench PRIVATE -O3)
if(TIEREDQ_TILES)
  target_compile_definitions(kernel_bench PRIVATE TIEREDQ_TILES)
endif()
target_link_libraries(kernel_bench PRIVATE TieredQ::TieredQ)
