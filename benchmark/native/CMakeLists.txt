# Linux CPU benchmark with identical optimization policy for both codecs.
cmake_minimum_required(VERSION 3.18)
project(codec_benchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(OpenMP REQUIRED)
add_compile_options(-O3 -march=native -ffp-contract=off)
set(TIEREDQ_TESTS OFF CACHE BOOL "" FORCE)
set(TIEREDQ_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(../.. tieredq)
find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
execute_process(COMMAND "${Python_EXECUTABLE}" -m nanobind --cmake_dir OUTPUT_VARIABLE NB_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
list(APPEND CMAKE_PREFIX_PATH "${NB_DIR}")
find_package(nanobind CONFIG REQUIRED)
option(BENCH_FULL_SCORE "Score all extra-bit codes without native bounds" OFF)
nanobind_add_module(codec_native NOMINSIZE module.cpp)
target_compile_definitions(codec_native PRIVATE BENCH_FULL_SCORE=$<BOOL:${BENCH_FULL_SCORE}>)
target_include_directories(codec_native PRIVATE "${RABITQ_SOURCE}/include")
target_link_libraries(codec_native PRIVATE TieredQ::TieredQ)
add_executable(codec_reference reference.cpp)
target_include_directories(codec_reference PRIVATE "${RABITQ_SOURCE}/include")
target_link_libraries(codec_reference PRIVATE OpenMP::OpenMP_CXX)
